# Portal Vida en Arte 360 — renombrar este archivo a .htaccess en el servidor
# Versión del 21-oct-2026 (rediseño): login y cambiar contraseña como páginas
# fijas, y la carpeta pública comun/ (estilos, capa de datos y logos).
Options -Indexes
DirectoryIndex index.html

RewriteEngine On

# Validación del certificado SSL (AutoSSL): debe quedar libre
RewriteRule ^\.well-known/ - [L]

# 1. Siempre por HTTPS
RewriteCond %{HTTPS} !=on
RewriteCond %{HTTP:X-Forwarded-Proto} !=https
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# 2. Archivos internos: nunca accesibles desde el navegador
RewriteRule ^(auth\.php|permisos\.php|cuentas\.php|usuarios\.php|php\.ini|htaccess\.txt)$ - [F,L]
RewriteRule (^|/)\. - [F,L]

# 3. Páginas que se abren directo, SIN pasar por el control de sesión.
#    Las .php que piden sesión la comprueban por dentro; login.html y
#    cambiar-clave.html son páginas fijas que no piden sesión (si la pidieran,
#    el login mandaría al login: bucle).
RewriteRule ^(login\.php|logout\.php|cambiar-clave\.php|acceso\.php|gate\.php|robots\.txt|login\.html|cambiar-clave\.html)$ - [L]

# 3c. comun/: estilos, capa de datos y logos. Públicos (el login los usa antes
#     de entrar), sin datos. Solo un nombre simple .css .js .png .svg
#     .webmanifest (el manifiesto de la app instalable), sin
#     subcarpetas; TODO lo demás dentro de comun/ (un .php, el listado de la
#     carpeta) se rechaza con 403. Va antes de 3b para que un comun/api.php
#     subido por error tampoco se ejecute.
RewriteRule ^comun/[a-z0-9-]+\.(css|js|png|svg|webmanifest)$ - [L]
RewriteRule ^comun(/|$) - [F,L]

# 3d. biblioteca/ (23-oct): presentaciones privadas, las entrega gate.php a
#     cualquiera con sesión. Ningún script se ejecuta ahí, aunque se llame
#     api.php (la regla 3b lo correría). Va antes de 3b.
RewriteRule ^biblioteca/.*\.(php|php[0-9]|phtml|phar)$ - [F,L]

# 3b. Los puentes de datos: comprueban sesión y permisos por dentro (requerir_sesion)
RewriteRule ^[a-z0-9-]+/(api|csv|config)\.php$ - [L]

# 4. Todo lo demás pasa por el control de sesión y de permisos (gate.php).
#    .well-known no llega aquí (regla de arriba); cgi-bin sí: como no está
#    asignada a ningún rol, solo la abren Administrador y Dirección.
RewriteRule ^(.*)$ gate.php?f=$1 [L,QSA]
